################################################################
#  実験2-b 左右の画面のビルドの設定（このファイルは「右」の exe の Setting フォルダにだけ置く）
#
#  ボタン（判断用スイッチ）をその試行で何回押したかを、画面の隅に小さく出す（実験者が見る用）。
#  正面の exe から UDP で回数が届く。「ボタン 0」「ボタン 1」「ボタン 2」と出る（3 回以上はオレンジ）。
#  「#」から右はメモ（読み込まれない）
################################################################

showButtonCount = 1        # 1 = 出す、0 = 出さない
buttonFontSize = 18        # 文字の大きさ（画面の高さ 1080 px のとき）
buttonDisplay = 2          # 出す画面の番号（この exe がつないでいる何番目の画面か。1, 2, 3 …）
buttonPosX = 0.97          # 画面の中の横の位置（0 = 左端、1 = 右端。文字の右端をここに合わせる）
buttonPosY = 0.25          # 画面の中の縦の位置（0 = 下端、1 = 上端）
showDisplayNumbers = 0     # 1 にすると、各画面の真ん中に「画面 1」「画面 2」… が大きく出る（何番か確かめるときだけ。本番は 0）
expectedPresses = 2        # この回数より多いとオレンジ（0〜2 回は白）
