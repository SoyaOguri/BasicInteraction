################################################################
#  実験2-b の設定ファイル（正面＝Main の PC だけで使う）
#
#  ■ 置き場所
#    正面の exe と同じフォルダの中の「Setting」フォルダに置く
#      例）Exp2b_Main\Exp2b.exe
#          Exp2b_Main\Setting\exp2b.txt   ← このファイル
#    ※ このファイルが見つからないと、Unity で設定した初期値
#      （被験者1・練習）で始まってしまうので注意
#
#  ■ 使い方（被験者が変わるたび・セットが変わるたび）
#    1. 下の participant と phase を書き換えて「上書き保存」
#    2. 左右の exe → 正面の exe の順に起動
#    3. 正面で Enter を押すと次の試行へ進む
#       起動して最初の試行だけは、実験者がテンキーの「＋」を押すと「3, 2, 1」が始まる
#       （2 試行目からは、評定に答え終わると自動で「3, 2, 1」が始まる）
#    4. 全部終わったら exe を閉じ、次の phase に書き換えて起動し直す
#
#  ■ 1人あたりの流れ
#    Set1（38 試行 = 練習 10 ＋ 本番 24 ＋ 車ありダミー 4）
#    Set2・Set3（各 29 試行 = 練習 1 ＋ 本番 24 ＋ 車ありダミー 4）
#    ・各セットの最初が練習（24 条件からランダム。練習が 2 回以上なら、そのうち 1 回は車が出る）
#      練習の回数は下の practiceCounts で変えられる
#    ・各試行のはじめに「3, 2, 1 → スタート」。0 になった瞬間に制限時間のタイマーが始まる
#      （カウントダウンの間は、こいでも自転車は進まない）
#    ・スタートしてから 3 秒間は、自動で 3 m/s で進む（止まった状態から走り出さないように）。そのあとは、こいだ速さ
#    ・ダミー 4 試行は、本番 24 試行の間のランダムな位置に入る（起動のたびに変わる）
#    ・独立の Practice（練習 3 試行・固定）も残っているが、普段は使わなくてよい
#
#  「#」から右はメモ（読み込まれない）。「=」の右だけ書き換える。
################################################################


# ---- 毎回書き換えるところ ------------------------------------

participant = 1        # 被験者番号：1〜8
phase = Set1           # Set1 / Set2 / Set3（Practice = 独立の練習。普段は使わない）


# ---- 途中で止まったときだけ書き換える ------------------------

startTrial = 1         # 何試行目から始めるか（普段は 1）
                       # 例）15試行目で落ちたら 15 にして起動し直す
                       #     終わったら 1 に戻すのを忘れずに
                       # ※ 途中から始めると、前回と同じ練習・ダミーの並びを自動で使う
                       #   （被験者フォルダの P01_Set1_seed.txt に残した乱数の種を読む）

seed = 0               # 練習とダミーの並びを決める乱数の種。0 = 起動のたびに新しく決める（普段は 0）
                       # 特定の並びを再現したいときだけ、seed.txt に残っている数字を書く


# ---- 校門までの制限時間（画面左下のカウントダウン） ----------
#  「3, 2, 1」が 0 になるとカウントダウンが始まり、校門（x = 1384.9）を通過できれば成功。
#  スタート（startBack = 10 なら x = 1265）から校門まで約 120 m。最初の 3 秒は自動で 3 m/s（約 9 m）なので
#    ・そのあと 5.5 m/s で走る → 約 23 秒（余裕 約 5 秒）
#    ・そのあと 5.0 m/s で走る → 約 25 秒（余裕 約 3 秒）
#    ・交差点で一時停止（減速→1秒停止→再加速） → 約 30 秒以上（間に合わない）
#  （以前は スタート x = 1275・動き出してから計測・25 秒）

timeLimit = 28

# 画面左下（タイマーの上）に「セット1　5 / 28 試行目」（練習中は「セット1　練習 2 / 3」）を出すか（1 = 出す / 0 = 出さない）
showProgress = 1

# 画面左下の文字の大きさ（画面の高さ 1080 px のときの大きさ。画面が大きければその割合で大きくなる）
timerFontSize = 20      # 校門通過制限時間まで 00:25
progressFontSize = 14   # セット1　5 / 28 試行目


# ---- データの保存先 ------------------------------------------
#  この中に被験者ごとのフォルダ（P01〜P08）が自動で作られ、そこにすべて入る
#    P03\P03_Set1_T05_F12_Stop_W5_Mirror_Color_CarL.csv   … 1試行の時系列（各行に条件の列つき。W5 = 塀の角が 5 m 手前）
#    P03\P03_ratings.csv                                  … 走行後の評定（テンキー。Set1。車の出る試行は出さない）
#    P03\P03_Set1_seed.txt                                … 練習とダミーの乱数の種と、その起動の試行の並び
#  評定の問いの文は、同じ Setting フォルダの exp2b_ratings.txt で変えられる
#    P03\P03_Set1_summary.csv                               … 1試行1行の一覧
#  ※ フォルダを作れないとき（D ドライブが無い PC など）は exe の横の Log\Exp2b に保存される

dataDir = D:\work\M1_Experiment\BasicInteraction_Experiment\Log\2-b


# ---- 車とぶつかったときの扱い ----------------------------------
#  自転車が車の外形に入ったら「衝突」として、画面中央に「衝突！」を出し、記録する
#    時系列の csv  … collision（その試行で衝突したら 1）、carGap（車とのすき間 m）
#    summary.csv   … collision、collisionSec（試行開始から何秒で衝突したか）、minCarGap（最も近づいたすき間 m）
#  車は自転車がこの距離（m）より近づくとブレーキをかけて止まる。
#  小さくするほど車は止まらなくなり、ぶつかりやすくなる（0 = ブレーキをかけない）。
#  消す（行ごと削除）と Unity で設定した値（8 m）のまま。

carSafeDistance = 8


# ---- 走行後の評定（テンキー） --------------------------------
ratePhases = Set1          # 交差点の評定 4 問を出すセット（「,」区切り。例 Set1,Set2）。空にすると出さない（車の出る試行では出さない）
timingPhases = Set1,Set2,Set3   # 「1 回目・2 回目のボタンを押したタイミング」の問いを出すセット。空にすると出さない
timingCarTrials = 1        # 1 = 車の出る試行（ダミー・練習の車あり）でもタイミングの問いを出す、0 = 出さない
                           # 答えは P04_ratings.csv の btnTiming（1 回目）・btnTiming2（2 回目）列に −2（2 秒早い）〜 0（ちょうど）〜 +2（2 秒遅い）で入る
                           # 問いを増やす前の評定ファイルがある被験者（P01〜P04）は、P04_ratings_2.csv のように新しいファイルになる

# ---- 練習の回数と、試行のはじめのカウントダウン ---------------
practiceCounts = 10,1,1    # 各セットの最初の練習の回数（セット1, セット2, セット3）
countdownSeconds = 3       # 「3, 2, 1」の秒数。0 にすると使わない（自転車が動き出したらタイマーが始まる、前のやり方）
autoStartSeconds = 3       # スタートしてから何秒、自動で走らせるか（0 = 使わない）
autoStartSpeed = 3         # そのあいだの速さ [m/s]。こいだ速さのほうが速ければ、そちらになる
startBack = 10             # スタート位置を、もとの位置（交差点の 75 m 手前）から何 m 後ろにするか。10 → 交差点の 85 m 手前
hudBottom = 160            # 正面の画面の左下（タイマーと「セット1　5 / 28 試行目」）を、下端から何 px 上に出すか（1080 px の画面のとき）。前は 40

# ---- 基本は触らない ------------------------------------------

baseOrder = FixedShuffle   # 試行順の元になる並び（FixedShuffle のまま）
carLeadSeconds = 1.0       # 車ありの試行で、車が自転車より何秒早く交差点に着くか

# 左右の PC へ交差点の条件を送る先（研究室の既存の通信と同じネットワーク）
syncAddress = 192.168.11.255
syncPort = 30101
